cmake_minimum_required(VERSION 3.25)
project(DustCreek LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/utf-8 /W3 /permissive-)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN SDL_MAIN_HANDLED)
find_package(SDL3 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(FFMPEG REQUIRED)
add_library(opening apps/player/game.cpp apps/player/tvgun.cpp)
target_include_directories(opening PUBLIC apps/player)
target_link_libraries(opening PUBLIC nlohmann_json::nlohmann_json ws2_32)
add_executable(DustCreek WIN32 apps/player/main.cpp apps/player/ui.cpp apps/player/media.cpp apps/player/sound.cpp)
target_include_directories(DustCreek PRIVATE ${FFMPEG_INCLUDE_DIRS})
target_link_directories(DustCreek PRIVATE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(DustCreek PRIVATE opening SDL3::SDL3 ${FFMPEG_LIBRARIES} bcrypt gdi32)
target_link_options(DustCreek PRIVATE /ENTRY:wmainCRTStartup)
include(CTest)
if(BUILD_TESTING)
  add_executable(opening_tests tests/opening.cpp)
  target_link_libraries(opening_tests PRIVATE opening)
  add_test(NAME opening_rules COMMAND opening_tests "${CMAKE_CURRENT_SOURCE_DIR}/worlds/western/opening")
endif()
install(TARGETS DustCreek RUNTIME DESTINATION .)
install(FILES docs/PLAY_FIRST_MINUTE.md DESTINATION . RENAME README.md)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
include(InstallRequiredSystemLibraries)
install(DIRECTORY worlds/western/opening DESTINATION worlds/western)
install(DIRECTORY assets/western/opening_slices DESTINATION assets/western FILES_MATCHING PATTERN "*.mp4")
install(DIRECTORY "${CMAKE_BINARY_DIR}/Release/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
foreach(port IN ITEMS sdl3 ffmpeg nlohmann-json)
  install(FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${port}/copyright" DESTINATION "licenses/${port}")
endforeach()

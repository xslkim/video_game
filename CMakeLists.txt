cmake_minimum_required(VERSION 3.25)
project(DustCreek LANGUAGES CXX)
option(BUILD_CHANGAN "Build and package the Changan prototype" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/utf-8 /W3 /permissive-)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN SDL_MAIN_HANDLED)
find_package(SDL3 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(FFMPEG REQUIRED)
add_library(opening apps/player/game.cpp apps/player/tvgun.cpp)
target_include_directories(opening PUBLIC apps/player)
target_link_libraries(opening PUBLIC nlohmann_json::nlohmann_json ws2_32)
add_library(player_media apps/player/ui.cpp apps/player/media.cpp apps/player/sound.cpp)
target_include_directories(player_media PUBLIC ${FFMPEG_INCLUDE_DIRS})
target_link_directories(player_media PUBLIC ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(player_media PUBLIC opening SDL3::SDL3 ${FFMPEG_LIBRARIES} gdi32)
add_executable(DustCreek WIN32 apps/player/main.cpp)
target_link_libraries(DustCreek PRIVATE player_media bcrypt)
target_link_options(DustCreek PRIVATE /ENTRY:wmainCRTStartup)
if(BUILD_CHANGAN)
  add_executable(Changan WIN32 apps/player/changan.cpp apps/player/changan_game.cpp apps/player/changan_story.cpp)
  target_link_libraries(Changan PRIVATE player_media bcrypt psapi)
  target_link_options(Changan PRIVATE /ENTRY:wmainCRTStartup)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(opening_tests tests/opening.cpp)
  target_link_libraries(opening_tests PRIVATE opening)
  add_test(NAME opening_rules COMMAND opening_tests "${CMAKE_CURRENT_SOURCE_DIR}/worlds/western/opening")
endif()
if(BUILD_CHANGAN)
  install(TARGETS Changan RUNTIME DESTINATION . COMPONENT GameRuntime)
  set(CHANGAN_LEVEL "m0" CACHE STRING "Changan encounter to package and launch")
  if(NOT CHANGAN_LEVEL MATCHES "^[a-z0-9_]+$")
    message(FATAL_ERROR "Invalid Changan level name")
  endif()
  string(TOUPPER "${CHANGAN_LEVEL}" changan_doc)
  install(FILES "docs/PLAY_CHANGAN_${changan_doc}.md" DESTINATION . RENAME README.md)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/changan-start.json" CONTENT "{\"level\":\"${CHANGAN_LEVEL}\"}\n")
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/changan-start.json" DESTINATION worlds/changan RENAME start.json)
  install(DIRECTORY "worlds/changan/${CHANGAN_LEVEL}" DESTINATION worlds/changan
    FILES_MATCHING PATTERN "level.json" PATTERN "*.mask" PATTERN "production" EXCLUDE)
  set(changan_level "${CMAKE_CURRENT_SOURCE_DIR}/worlds/changan/${CHANGAN_LEVEL}/level.json")
  if(NOT CHANGAN_LEVEL STREQUAL "m0")
    install(DIRECTORY "assets/changan/audio/foley" DESTINATION assets/changan/audio FILES_MATCHING PATTERN "*.wav")
    install(FILES "assets/changan/audio/music/festival.wav" DESTINATION assets/changan/audio/music)
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${changan_level}")
  file(READ "${changan_level}" changan_metadata)
  string(JSON street_type ERROR_VARIABLE street_error TYPE "${changan_metadata}" street)
  if(street_type STREQUAL "OBJECT")
    install(DIRECTORY "assets/changan/${CHANGAN_LEVEL}/once" DESTINATION "assets/changan/${CHANGAN_LEVEL}"
      FILES_MATCHING PATTERN "*.mp4" PATTERN "*.mkv")
  else()
    install(DIRECTORY "assets/changan/${CHANGAN_LEVEL}" DESTINATION assets/changan
      FILES_MATCHING PATTERN "*.mp4" PATTERN "once" EXCLUDE)
  endif()
else()
  install(TARGETS DustCreek RUNTIME DESTINATION .)
  install(FILES docs/PLAY_FIRST_MINUTE.md DESTINATION . RENAME README.md)
  install(DIRECTORY worlds/western/opening DESTINATION worlds/western)
  install(DIRECTORY assets/western/opening_slices DESTINATION assets/western FILES_MATCHING PATTERN "*.mp4")
endif()
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT GameRuntime)
include(InstallRequiredSystemLibraries)
install(DIRECTORY "${CMAKE_BINARY_DIR}/Release/" DESTINATION . COMPONENT GameRuntime FILES_MATCHING PATTERN "*.dll")
foreach(port IN ITEMS sdl3 ffmpeg nlohmann-json)
  install(FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${port}/copyright" DESTINATION "licenses/${port}" COMPONENT GameRuntime)
endforeach()
